## **(2) 얼굴 데이터 수집 스크립트 개발**

- Python으로 `capture_my_face.py` 작성 (사진 찍기 위해서)
    
    ```
    import os
    import cv2
    import numpy as np
    import requests
    from datetime import datetime
    
    # 📌 네 ESP32-CAM IP 주소로 수정
    ESP_URL = "http://192.168.1.38/jpg"
    
    BASE_DIR = "dataset_me"
    os.makedirs(os.path.join(BASE_DIR, "me"), exist_ok=True)
    os.makedirs(os.path.join(BASE_DIR, "not_me"), exist_ok=True)
    
    def fetch_frame():
        try:
            resp = requests.get(ESP_URL, timeout=5)
            arr = np.frombuffer(resp.content, dtype=np.uint8)
            img = cv2.imdecode(arr, cv2.IMREAD_COLOR)
            return img
        except:
            return None
    
    print("=== 데이터 수집 ===")
    print("m = ME(나)  /  n = NOT ME(배경/다른사람) / q = 종료")
    print("=============================================")
    
    while True:
        frame = fetch_frame()
        if frame is None:
            continue
    
        # ---------- 여기서부터 가이드 박스 그리기 ----------
        disp = frame.copy()          # 화면에만 쓸 복사본
    
        h, w, _ = disp.shape
        box_w = int(w * 0.5)         # 가이드 박스 가로 비율
        box_h = int(h * 0.6)         # 가이드 박스 세로 비율
    
        cx, cy = w // 2, h // 2      # 화면 중앙
        x1 = cx - box_w // 2
        y1 = cy - box_h // 2
        x2 = cx + box_w // 2
        y2 = cy + box_h // 2
    
        # 초록색 박스
        cv2.rectangle(disp, (x1, y1), (x2, y2), (0, 255, 0), 2)
        # 안내 텍스트
        cv2.putText(
            disp,
            "Put your face inside the box (m=ME, n=NOT ME, q=quit)",
            (10, 20),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.5,
            (0, 255, 0),
            1,
        )
        
        disp_large = cv2.resize(disp, (900, 700)) 
        # ---------- 여기까지 오버레이 ----------
    
        cv2.imshow("ESP32 Feed", disp_large)
        key = cv2.waitKey(1) & 0xFF
    
        if key == ord('q'):
            break
    
        ts = datetime.now().strftime("%Y%m%d_%H%M%S_%f")
    
        # 저장되는 이미지는 오버레이 없는 '원본 frame' 사용!
        if key == ord('m'):
            path = f"{BASE_DIR}/me/{ts}.jpg"
            cv2.imwrite(path, frame)
            print("Saved → ME      :", path)
    
        elif key == ord('n'):
            path = f"{BASE_DIR}/not_me/{ts}.jpg"
            cv2.imwrite(path, frame)
            print("Saved → NOT_ME  :", path)
    
    cv2.destroyAllWindows()
    ```

## **(3) PC에서 TinyML 학습 파이프라인 구축**

- `train_me_notme.py` 작성
    
    ```python
    import os
    import numpy as np
    import cv2
    from sklearn.model_selection import train_test_split
    import tensorflow as tf
    from tensorflow import keras
    from tensorflow.keras import layers
    
    # 데이터 경로 설정
    DATA_DIR_ME = "./dataset_me/me"
    DATA_DIR_NOT_ME = "./dataset_me/not_me"
    
    IMG_SIZE = 64
    
    # 이미지 로딩
    # 이미지 한 장씩: 
        # Grayscale로 읽기
        # 64x64로 resize
        # 0~1 정규화
        # numpy 배열에 추가
        # 각 이미지에 라벨(1=me, 0=not me) 붙임
    # 이미지 반환:
        # images: 처리된 이미지 목록
        # labels: 각 이미지의 라벨 목록
    def load_images_from_folder(folder, label):
        images = []
        labels = []
        for filename in os.listdir(folder):
            path = os.path.join(folder, filename)
            img = cv2.imread(path, cv2.IMREAD_GRAYSCALE)  # grayscale
            if img is None:
                continue
            img = cv2.resize(img, (IMG_SIZE, IMG_SIZE))
            img = img.astype("float32") / 255.0
            images.append(img)
            labels.append(label)
        return images, labels
    
    # 데이터셋 구축
    def load_dataset():
        me_imgs, me_labels = load_images_from_folder(DATA_DIR_ME, 1)
        not_imgs, not_labels = load_images_from_folder(DATA_DIR_NOT_ME, 0)
    
        X = np.array(me_imgs + not_imgs)
        y = np.array(me_labels + not_labels)
    
        # reshape: (n, 64, 64) → (n, 4096) -> Dense 모델 입력으로 만들기 위해
        X = X.reshape((X.shape[0], IMG_SIZE * IMG_SIZE))
    
        # 80% 학습용, 20% 검증용
        return train_test_split(X, y, test_size=0.2, stratify=y, random_state=42)
    
    # Dense Neural Network 모델 구성
    # Dense(64) -> Dense(32) -> Dense(1)
    def build_dense_model():
        model = keras.Sequential([
            layers.Input(shape=(IMG_SIZE * IMG_SIZE,)),
            layers.Dense(64, activation="relu"),
            layers.Dense(32, activation="relu"),
            layers.Dense(1, activation="sigmoid")  # me or not_me
        ])
    
        model.compile(
            optimizer="adam",
            loss="binary_crossentropy",
            metrics=["accuracy"]
        )
    
        model.summary()
        return model
    
    def main():
        X_train, X_val, y_train, y_val = load_dataset()
        print("Train:", X_train.shape, y_train.shape)
        print("Val  :", X_val.shape, y_val.shape)
    
        model = build_dense_model()
        # 모델 학습
        model.fit(X_train, y_train, epochs=30, batch_size=8,
                  validation_data=(X_val, y_val))
    
        # Save Keras
        keras_path = "./model_out/me_notme_dense.h5"
        model.save(keras_path)
        print("💾 Saved Keras model:", keras_path)
    
        # Convert to TFLite
        converter = tf.lite.TFLiteConverter.from_keras_model(model)
        converter.optimizations = [tf.lite.Optimize.DEFAULT]  # quantization friendly
        tflite_model = converter.convert()
    
        tflite_path = "./model_out/me_notme_dense.tflite"
        with open(tflite_path, "wb") as f:
            f.write(tflite_model)
    
        print("💾 Saved TFLite model:", tflite_path)
        print("🎉 Done.")
    
    if __name__ == "__main__":
        main()
    ```

## **(4) ESP32에 올리기 위한 변환**

- TFLite 파일을 C 배열로 변환 → Python 변환(`tflite_to_c_array.py`)
    
    ```python
    from pathlib import Path
    
    # 1) 입력 tflite 파일 경로
    #    -> 지금 구조 기준: C:\WS\face_recognition\src\model_out\me_notme_model.tflite
    project_root = Path(__file__).parent
    tflite_path = project_root / "src" / "model_out" / "me_notme_model.tflite"
    
    # 2) 출력 헤더 파일 경로
    header_path = project_root / "src" / "model_data.h"
    
    array_name = "g_me_notme_model"
    
    def main():
        if not tflite_path.exists():
            raise FileNotFoundError(f"tflite 파일을 찾을 수 없음: {tflite_path}")
    
        data = tflite_path.read_bytes()
        print(f"[*] Read {len(data)} bytes from {tflite_path}")
    
        with header_path.open("w", encoding="utf-8") as f:
            f.write("#pragma once\n\n")
            f.write("// Auto-generated from me_notme_model.tflite\n")
            f.write(f"const unsigned char {array_name}[] = {{\n  ")
    
            for i, b in enumerate(data):
                f.write(f"0x{b:02x}, ")
                # 1줄에 12개씩 끊기
                if (i + 1) % 12 == 0:
                    f.write("\n  ")
    
            f.write("\n};\n\n")
            f.write(f"const unsigned int {array_name}_len = {len(data)};\n")
    
        print(f"[*] Wrote C array to {header_path}")
    
    if __name__ == "__main__":
        main()
    ```
    
- `model_data.h` 생성

## **(5) ESP32-S3에 TFLite Micro + EloquentTinyML 적용 시도**

- `main.cpp`에 TinyML 모델 로딩 코드 추가
- 브라우저 실시간 스트림 + TinyML 모델 초기화까지 통합
- 여기서부터 문제가 발생…
    1. **TFLite Micro 오퍼레이터 버전 충돌**
    
    로그:
    
    ```
    Didn't find op for builtin opcode 'CONV_2D' version '5'
    Didn't find op for builtin opcode 'FULLY_CONNECTED' version '9'
    AllocateTensors() failed
    ```
    
    - EloquentTinyML은 **TensorFlow Lite Micro의 매우 제한된 오퍼레이터 세트**만 지원
    - 우리가 만든 CNN 모델은 아래 연산 포함:
        - Conv2D
        - MaxPooling
        - Dense(v9)
        - Dropout
        - Flatten
    
    → ESP32에서 지원하지 않는 연산 버전 사용
    
    → 모델 로딩 자체가 실패
    
    1. 연산량 부족
    
    CNN이나 Dense 모델을 64×64 입력으로 돌릴 경우:
    
    - ESP32-S3의 240MHz CPU로는 매우 무거움
    - 초당 1–2프레임도 어려움
    - 실시간 인식 불가능
