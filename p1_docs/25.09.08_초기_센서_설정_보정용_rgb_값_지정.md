## 초기 센서 설정

---

- 센서-표면 10mm 설정
- 기존 코드 문제점: rgb 값이 0~255 범위가 아니였음
    - 센서가 16-bit RAW 값을 그대로 반환하였음
    - 최대 65535를 뱉음
- 수정 코드: 16-bit 값을 8-bit 값으로 변환하여 출력
- colorsensor.cpp 코드

```cpp
#include <Arduino.h>
#include <Wire.h>
#include "Adafruit_TCS34725.h"
#include "colorsensor.h"

   
// 핀 매핑
//  - 현재 값: GPIO4(D2)=SDA, GPIO5(D3)=SCL
#define SDA_PIN 4
#define SCL_PIN 5

// TCS34725 설정: 적분시간/이득
//  - 밝을수록: 적분시간↓, 이득↓
//  - 어두울수록: 적분시간↑, 이득↑
static Adafruit_TCS34725 tcs(
  TCS34725_INTEGRATIONTIME_154MS,
  TCS34725_GAIN_1X
);

void initColorSensor() {
  // I2C 시작
  Wire.begin(SDA_PIN, SCL_PIN);

  if (!tcs.begin()) {
    Serial.println("[TCS34725] Not found! Check wiring.");
    while (1) { delay(1000); }
  }
  Serial.println("[TCS34725] Found.");
}

void readRawRGB(int &rOut, int &gOut, int &bOut) {
  uint16_t r, g, b, c;  // 센서는 16-bit RAW(최대 65535)를 뱉음
  tcs.getRawData(&r, &g, &b, &c);

  // 그대로 반환 (0~65535)
  rOut = (int)r;
  gOut = (int)g;
  bOut = (int)b;

  // 디버그 출력
  Serial.print("[RAW] R: "); Serial.print(rOut);
  Serial.print(" G: ");    Serial.print(gOut);
  Serial.print(" B: ");    Serial.print(bOut);
  Serial.print(" C: ");    Serial.println((int)c);

  // 과/저노출 힌트
  if (c > 64000) {  // 포화 위험 -> 적분시간/Gain 낮추기
    Serial.println("[TCS34725] ⚠ C near saturation. Reduce gain or integration time.");
  } else if (c < 10) {  // 너무 어두움 -> 적분시간/Gain 높이기
    Serial.println("[TCS34725] ℹ Very low light. Consider higher gain/time or better shielding.");
  }
}

//8-bit 변환
void readRgb8ForBle(uint8_t &r8, uint8_t &g8, uint8_t &b8, uint16_t &cOut) {
  uint16_t r, g, b, c;
  tcs.getRawData(&r, &g, &b, &c);
  cOut = c;

  if (c == 0) {
    // 완전 암부 노이즈 회피
    r8 = g8 = b8 = 0;
    return;
  }

  // 밝기 보정: 채널을 Clear로 나누어 0~1 근사
  float rn = (float)r / (float)c;
  float gn = (float)g / (float)c;
  float bn = (float)b / (float)c;

  // 대비 확보: 최댓값을 1로 맞춤
  float maxn = rn;
  if (gn > maxn) maxn = gn;
  if (bn > maxn) maxn = bn;
  if (maxn < 1e-6f) {
    r8 = g8 = b8 = 0;
    return;
  }

  float r01 = rn / maxn;
  float g01 = gn / maxn;
  float b01 = bn / maxn;

  
  int ri = (int)(r01 * 255.0f + 0.5f);
  int gi = (int)(g01 * 255.0f + 0.5f);
  int bi = (int)(b01 * 255.0f + 0.5f);
  if (ri < 0) ri = 0; if (ri > 255) ri = 255;
  if (gi < 0) gi = 0; if (gi > 255) gi = 255;
  if (bi < 0) bi = 0; if (bi > 255) bi = 255;
  r8 = (uint8_t)ri;
  g8 = (uint8_t)gi;
  b8 = (uint8_t)bi;

}

```

### **결론 & 개선 포인트**

1. **흰/검/빨/노** → 안정적
2. **파랑** → 조금 어둡게 나옴 → LED 밝기 혹은 통합 광량 보정 필요
3. **초록** → 거의 읽히지 않음 → 센서 민감도 부족, 캘리브레이션 필요

💡 **추천 조치**

- 초록 계열과 파랑 계열은 **화이트 캘리브레이션 후 색상별 보정**을 강화
- 센서와 LED 간 거리, 광원 강도, 주변 빛 차단 → 측정 정확도 개선
- 필요하면 **보정용 CCM(컬러 교정 행렬)** 적용 후 다시 테스트
